-- 1. Create Database if not exists
IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'SPLIT_BILL_DB')
BEGIN
    CREATE DATABASE [SPLIT_BILL_DB];
END
GO

USE [SPLIT_BILL_DB];
GO

-- ==========================================================
-- STEP 1: CLEAN / WIPE ALL EXISTING TABLES (RESET SCRIPT)
-- ==========================================================
IF OBJECT_ID('dbo.ActivityLogs', 'U') IS NOT NULL DROP TABLE dbo.ActivityLogs;
IF OBJECT_ID('dbo.Notifications', 'U') IS NOT NULL DROP TABLE dbo.Notifications;
IF OBJECT_ID('dbo.TripCollaborators', 'U') IS NOT NULL DROP TABLE dbo.TripCollaborators;
IF OBJECT_ID('dbo.FriendConnections', 'U') IS NOT NULL DROP TABLE dbo.FriendConnections;
IF OBJECT_ID('dbo.SettlementTransactions', 'U') IS NOT NULL DROP TABLE dbo.SettlementTransactions;
IF OBJECT_ID('dbo.ExpenseSplits', 'U') IS NOT NULL DROP TABLE dbo.ExpenseSplits;
IF OBJECT_ID('dbo.Expenses', 'U') IS NOT NULL DROP TABLE dbo.Expenses;
IF OBJECT_ID('dbo.Persons', 'U') IS NOT NULL DROP TABLE dbo.Persons;
IF OBJECT_ID('dbo.Groups', 'U') IS NOT NULL DROP TABLE dbo.Groups;
IF OBJECT_ID('dbo.Trips', 'U') IS NOT NULL DROP TABLE dbo.Trips;
IF OBJECT_ID('dbo.OtpVerification', 'U') IS NOT NULL DROP TABLE dbo.OtpVerification;
IF OBJECT_ID('dbo.UserTokens', 'U') IS NOT NULL DROP TABLE dbo.UserTokens;
IF OBJECT_ID('dbo.Users', 'U') IS NOT NULL DROP TABLE dbo.Users;
GO

-- ==========================================================
-- STEP 2: CREATE ALL TABLES (STANDALONE / NO FOREIGN KEYS)
-- ==========================================================

-- 1. USERS TABLE
CREATE TABLE Users (
    user_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    phone_number NVARCHAR(20) NOT NULL,
    full_name NVARCHAR(150) NULL,
    password_hash NVARCHAR(255) NULL,
    auth_token NVARCHAR(500) NULL,
    token_expiry DATETIME NULL,
    diet_type INT DEFAULT 0, -- 0: veg, 1: non-veg, 2: non-veg + alc
    diet_name NVARCHAR(50) DEFAULT 'Vegetarian',
    avatar_color NVARCHAR(20) DEFAULT '#4F46E5',
    is_biometric_enabled BIT DEFAULT 0,
    is_active BIT DEFAULT 1,
    created_at DATETIME DEFAULT GETDATE(),
    updated_at DATETIME DEFAULT GETDATE(),
    last_login_at DATETIME NULL
);
CREATE INDEX IX_Users_Phone ON Users(phone_number);
GO

-- 2. USER TOKENS / SESSIONS TABLE
CREATE TABLE UserTokens (
    token_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    user_id NVARCHAR(100) NOT NULL,
    phone_number NVARCHAR(20) NULL,
    auth_token NVARCHAR(500) NOT NULL,
    device_info NVARCHAR(255) NULL,
    is_active BIT DEFAULT 1,
    expires_at DATETIME NULL,
    created_at DATETIME DEFAULT GETDATE(),
    last_used_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_UserTokens_AuthToken ON UserTokens(auth_token);
CREATE INDEX IX_UserTokens_UserId ON UserTokens(user_id);
GO

-- 3. OTP VERIFICATION TABLE
CREATE TABLE OtpVerification (
    otp_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    phone_number NVARCHAR(20) NOT NULL,
    otp_code NVARCHAR(10) NOT NULL,
    purpose NVARCHAR(50) DEFAULT 'REGISTER', -- 'REGISTER', 'FORGOT_PASSWORD', 'LOGIN'
    is_verified BIT DEFAULT 0,
    expires_at DATETIME NOT NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Otp_Phone_Purpose ON OtpVerification(phone_number, purpose);
GO

-- 4. TRIPS TABLE
CREATE TABLE Trips (
    trip_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    user_id NVARCHAR(100) NOT NULL,
    name NVARCHAR(200) NOT NULL,
    description NVARCHAR(MAX) NULL,
    currency_code NVARCHAR(10) DEFAULT 'INR',
    created_at DATETIME DEFAULT GETDATE(),
    updated_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Trips_UserId ON Trips(user_id);
GO

-- 5. GROUPS TABLE
CREATE TABLE Groups (
    group_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    trip_id NVARCHAR(100) NOT NULL,
    name NVARCHAR(200) NOT NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Groups_TripId ON Groups(trip_id);
GO

-- 6. PERSONS / MEMBERS TABLE
CREATE TABLE Persons (
    person_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    group_id NVARCHAR(100) NULL,
    trip_id NVARCHAR(100) NOT NULL,
    user_id NVARCHAR(100) NULL,
    phone_number NVARCHAR(20) NULL,
    name NVARCHAR(150) NOT NULL,
    diet_type INT DEFAULT 0,
    diet_name NVARCHAR(50) DEFAULT 'Vegetarian',
    paid_amount DECIMAL(18, 2) DEFAULT 0.00,
    owed_amount DECIMAL(18, 2) DEFAULT 0.00,
    balance DECIMAL(18, 2) DEFAULT 0.00,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Persons_TripId ON Persons(trip_id);
CREATE INDEX IX_Persons_UserId ON Persons(user_id);
CREATE INDEX IX_Persons_Phone ON Persons(phone_number);
GO

-- 7. EXPENSES TABLE
CREATE TABLE Expenses (
    expense_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    trip_id NVARCHAR(100) NOT NULL,
    description NVARCHAR(500) NOT NULL,
    amount DECIMAL(18, 2) NOT NULL,
    currency_code NVARCHAR(10) DEFAULT 'INR',
    category_index INT DEFAULT 0,
    category_name NVARCHAR(100) DEFAULT 'General',
    paid_by_person_id NVARCHAR(100) NOT NULL,
    split_type NVARCHAR(50) DEFAULT 'EQUAL', -- 'EQUAL', 'EXACT', 'DIETARY'
    expense_date DATETIME NOT NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Expenses_TripId ON Expenses(trip_id);
CREATE INDEX IX_Expenses_PaidBy ON Expenses(paid_by_person_id);
GO

-- 8. EXPENSE SPLITS TABLE
CREATE TABLE ExpenseSplits (
    split_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    expense_id NVARCHAR(100) NOT NULL,
    trip_id NVARCHAR(100) NOT NULL,
    person_id NVARCHAR(100) NOT NULL,
    amount DECIMAL(18, 2) NOT NULL DEFAULT 0.00,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_ExpenseSplits_ExpenseId ON ExpenseSplits(expense_id);
CREATE INDEX IX_ExpenseSplits_TripId ON ExpenseSplits(trip_id);
CREATE INDEX IX_ExpenseSplits_PersonId ON ExpenseSplits(person_id);
GO

-- 9. SETTLEMENT TRANSACTIONS TABLE
CREATE TABLE SettlementTransactions (
    settlement_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    trip_id NVARCHAR(100) NOT NULL,
    from_person_id NVARCHAR(100) NOT NULL,
    to_person_id NVARCHAR(100) NOT NULL,
    amount DECIMAL(18, 2) NOT NULL DEFAULT 0.00,
    is_completed BIT DEFAULT 0,
    status NVARCHAR(50) DEFAULT 'ACCEPTED', -- 'PENDING', 'ACCEPTED', 'DECLINED'
    created_by_user_id NVARCHAR(100) NULL,
    settled_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Settlements_TripId ON SettlementTransactions(trip_id);
CREATE INDEX IX_Settlements_Status ON SettlementTransactions(status);
GO

-- 10. FRIEND CONNECTIONS TABLE
CREATE TABLE FriendConnections (
    connection_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    user_id NVARCHAR(100) NOT NULL,
    friend_user_id NVARCHAR(100) NULL,
    friend_phone NVARCHAR(20) NOT NULL,
    friend_name NVARCHAR(150) NOT NULL,
    diet_type INT DEFAULT 0,
    diet_name NVARCHAR(50) DEFAULT 'Vegetarian',
    status NVARCHAR(50) DEFAULT 'ACCEPTED', -- 'PENDING', 'ACCEPTED', 'REJECTED'
    requester_id NVARCHAR(100) NULL,
    receiver_id NVARCHAR(100) NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Friends_UserId ON FriendConnections(user_id);
CREATE INDEX IX_Friends_Phone ON FriendConnections(friend_phone);
CREATE INDEX IX_Friends_FriendUserId ON FriendConnections(friend_user_id);
GO

-- 11. TRIP COLLABORATORS TABLE
CREATE TABLE TripCollaborators (
    collab_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    trip_id NVARCHAR(100) NOT NULL,
    user_id NVARCHAR(100) NULL,
    person_id NVARCHAR(100) NULL,
    phone_number NVARCHAR(20) NULL,
    role NVARCHAR(50) DEFAULT 'MEMBER', -- 'OWNER', 'MEMBER', 'VIEWER'
    status NVARCHAR(50) DEFAULT 'ACCEPTED', -- 'PENDING', 'ACCEPTED'
    joined_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Collaborators_TripId ON TripCollaborators(trip_id);
CREATE INDEX IX_Collaborators_UserId ON TripCollaborators(user_id);
CREATE INDEX IX_Collaborators_Phone ON TripCollaborators(phone_number);
GO

-- 12. NOTIFICATIONS TABLE
CREATE TABLE Notifications (
    notification_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    user_id NVARCHAR(100) NOT NULL,
    trip_id NVARCHAR(100) NULL,
    trip_name NVARCHAR(200) NULL,
    title NVARCHAR(200) NOT NULL,
    message NVARCHAR(MAX) NOT NULL,
    type NVARCHAR(50) NOT NULL, -- 'EXPENSE_ADDED', 'EXPENSE_UPDATED', 'DEBT_ALERT', 'SETTLEMENT', 'COLLAB_INVITE'
    amount DECIMAL(18, 2) DEFAULT 0.00,
    is_read BIT DEFAULT 0,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_Notifications_UserId ON Notifications(user_id, is_read);
GO

-- 13. ACTIVITY LOGS TABLE
CREATE TABLE ActivityLogs (
    log_id NVARCHAR(100) NOT NULL PRIMARY KEY,
    user_id NVARCHAR(100) NULL,
    trip_id NVARCHAR(100) NULL,
    action_type NVARCHAR(100) NOT NULL,
    details NVARCHAR(MAX) NULL,
    created_at DATETIME DEFAULT GETDATE()
);
CREATE INDEX IX_ActivityLogs_TripId ON ActivityLogs(trip_id);
CREATE INDEX IX_ActivityLogs_UserId ON ActivityLogs(user_id);
GO
