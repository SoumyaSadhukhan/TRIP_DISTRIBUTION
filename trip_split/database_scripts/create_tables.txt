-- 1. Create Database if it does not exist
IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'SPLIT_BILL_DB')
BEGIN
    CREATE DATABASE [SPLIT_BILL_DB];
END
GO

USE [SPLIT_BILL_DB];
GO

-- ==========================================================
-- UNIT 1: USERS TABLE
-- Stores user core profile, credentials, dietary preferences, and token cache info
-- ==========================================================
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'Users')
BEGIN
    CREATE TABLE Users (
        user_id NVARCHAR(100),
        phone_number NVARCHAR(20),
        full_name NVARCHAR(150),
        password_hash NVARCHAR(255),
        auth_token NVARCHAR(500),
        token_expiry DATETIME,
        diet_type INT DEFAULT 0, -- 0: vegetarian, 1: nonVegetarian, 2: nonVegAlcoholic
        diet_name NVARCHAR(50) DEFAULT 'Vegetarian',
        avatar_color NVARCHAR(20) DEFAULT '#4F46E5',
        is_biometric_enabled BIT DEFAULT 0,
        is_active BIT DEFAULT 1,
        created_at DATETIME DEFAULT GETDATE(),
        updated_at DATETIME DEFAULT GETDATE(),
        last_login_at DATETIME
    );
END
ELSE
BEGIN
    -- Add columns if missing in existing table
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('Users') AND name = 'diet_type')
        ALTER TABLE Users ADD diet_type INT DEFAULT 0;
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('Users') AND name = 'diet_name')
        ALTER TABLE Users ADD diet_name NVARCHAR(50) DEFAULT 'Vegetarian';
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('Users') AND name = 'avatar_color')
        ALTER TABLE Users ADD avatar_color NVARCHAR(20) DEFAULT '#4F46E5';
END
GO

-- ==========================================================
-- UNIT 2: USER TOKENS / SESSIONS TABLE
-- Stores persistent login tokens for auto-login & multi-device sessions
-- ==========================================================
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'UserTokens')
BEGIN
    CREATE TABLE UserTokens (
        token_id NVARCHAR(100),
        user_id NVARCHAR(100),
        phone_number NVARCHAR(20),
        auth_token NVARCHAR(500),
        device_info NVARCHAR(255),
        is_active BIT DEFAULT 1,
        expires_at DATETIME,
        created_at DATETIME DEFAULT GETDATE(),
        last_used_at DATETIME DEFAULT GETDATE()
    );
END
GO

-- ==========================================================

GO

-- ==========================================================

-- ==========================================================
-- UNIT 6: PERSONS / MEMBERS TABLE
-- Stores individual participants, dietary types, and balance states
-- ==========================================================
IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'Persons')
BEGIN
    CREATE TABLE Persons (
        person_id NVARCHAR(100),
        group_id NVARCHAR(100),
        trip_id NVARCHAR(100),
        user_id NVARCHAR(100), -- Linked user account if registered
        phone_number NVARCHAR(20),
        name NVARCHAR(150),
        diet_type INT, -- 0: vegetarian, 1: nonVegetarian, 2: nonVegAlcoholic
        diet_name NVARCHAR(50),
        paid_amount DECIMAL(18, 2) DEFAULT 0.00,
        owed_amount DECIMAL(18, 2) DEFAULT 0.00,
        balance DECIMAL(18, 2) DEFAULT 0.00,
        created_at DATETIME DEFAULT GETDATE()
    );
END
ELSE
BEGIN
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('Persons') AND name = 'user_id')
        ALTER TABLE Persons ADD user_id NVARCHAR(100);
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('Persons') AND name = 'phone_number')
        ALTER TABLE Persons ADD phone_number NVARCHAR(20);
END
GO

-- ==========================================================
-- UNIT 9: SETTLEMENT TRANSACTIONS TABLE

-- ==========================================================

END
ELSE

        ALTER TABLE SettlementTransactions ADD created_by_user_id NVARCHAR(100);
END
GO
